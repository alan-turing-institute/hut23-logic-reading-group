root ::= ("\n")* (line "\n")*
line ::= step ":" space rule
step ::= [1-9][0-9]*
rule ::= premise | reiteration | and-intro | and-elim-left | and-elim-right | imp-elim | imp-intro | or-intro-left | or-intro-right | or-elim | not-elim | not-intro | indirect | explosion | assumption | discharge | modus-tollens | double-not-elim
ref ::= [1-9][0-9]*
exp ::= "$" outer "$"
outer ::= var | "\\\\neg " wff | wff space connective space wff | wff
wff ::= var | "\\\\neg " wff | "(" wff space connective space wff ")"
connective ::= "\\\\lor" | "\\\\land" | "\\\\to"
var ::= [a-zA-Z]+
space ::= " "
premise ::= "premise" space exp
reiteration ::= "reiteration" space ref
and-intro ::= "and_intro" space ref "," space ref
and-elim-left ::= "and_elim_left" space ref
and-elim-right ::= "and_elim_right" space ref
imp-elim ::= "imp_elim" space ref "," space ref
imp-intro ::= "imp_intro" space ref "," space ref
or-intro-left ::= "or_intro_left" space ref "," space exp
or-intro-right ::= "or_intro_right" space ref "," space exp
or-elim ::= "or_elim" space ref "," space ref "," space ref "," space ref "," space ref
not-elim ::= "not_elim" space ref "," space ref
not-intro ::= "not_intro" space ref "," space ref
indirect ::= "indirect" space ref "," space ref
explosion ::= "explosion" space ref "," space exp
assumption ::= "assumption" space exp
discharge ::= "discharge"
modus-tollens ::= "modus_tollens" space ref "," space ref
double-not-elim ::= "double_not_elim" space ref
